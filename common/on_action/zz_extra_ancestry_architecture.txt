# Extra Ancestry Architecture: birth handler for the traits vanilla does not know about.
#
# Vanilla reads exactly eight dynasty variables (blood_legacy_4_<trait>) in on_birth_child
# (common/on_action/child_birth_on_actions.txt). Picking any other trait in the perk window therefore
# did nothing. This on_action reads the variable for each extra trait and applies it with the same rules
# as vanilla: the child's dynasty has the perk, a real parent is of that dynasty, the child does not
# already carry a conflicting trait, and the chance is blood_legacy_4_chance (+ blood_legacy_4_extra_player_chance
# when a parent is a player). Fecund, Giant, Dwarf, Albino and Scaly stay with vanilla.
on_birth_child = {
	on_actions = { extra_ancestry_architecture_birth }
}

extra_ancestry_architecture_birth = {
	effect = {
		hidden_effect = {
			if = {
				limit = {
					exists = scope:child.dynasty
					scope:child.dynasty = { has_dynasty_perk = blood_legacy_4 }
					OR = { # The child must have a real parent of the Dynasty...
						scope:mother.dynasty = scope:child.dynasty
						scope:real_father.dynasty = scope:child.dynasty
					}
				}
				if = {
					limit = {
						scope:child.dynasty = { exists = var:blood_legacy_4_beauty_good_3 }
						NOR = { has_trait = beauty_good has_trait = beauty_bad }
					}
					random = {
						chance = blood_legacy_4_chance
						modifier = {
							add = blood_legacy_4_extra_player_chance
							any_parent = { is_ai = no }
						}
						add_trait = beauty_good_3
					}
				}
				else_if = {
					limit = {
						scope:child.dynasty = { exists = var:blood_legacy_4_intellect_good_3 }
